#include <bits/stdc++.h>
using namespace std;

string timeConversion(string s) {
    int hour = stoi(s.substr(0, 2));
    string period = s.substr(8, 2);

    if (period == "AM" && hour == 12) {
        hour = 0;
    }
    else if (period == "PM" && hour != 12) {
        hour += 12;
    }

    string result = s.substr(0, 8);

    // Replace the hour with the converted hour
    result[0] = '0' + hour / 10;
    result[1] = '0' + hour % 10;

    return result;
}
